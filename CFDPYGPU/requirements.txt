# ---------------------------------------------------------------------------

# ---------------------------------------------------------------------------
# Core runtime (required).  All three are hard dependencies: the framework
# uses NumPy array numerics, SciPy sparse matrices / Krylov solvers / ILU, and
# Matplotlib for PNG plots and MP4/GIF animations.

scipy>=1.11
matplotlib>=3.7

# ---------------------------------------------------------------------------
# Recommended extras (used by the default examples; the framework degrades

numba>=0.58            # @njit TVD flux-limiter kernel (pure-NumPy fallback)
                       # AND the @cuda.jit GPU kernels in gpu/ (needs a CUDA
                       # build of numba + an NVIDIA driver; falls back to CPU)
